import gradio as gr import torch from diffusers import StableDiffusionPipeline MODEL_ID = "segmind/tiny-sd" pipe = None def load_model(): global pipe if pipe is None: pipe = StableDiffusionPipeline.from_pretrained( MODEL_ID, torch_dtype=torch.float32 ) pipe = pipe.to("cpu") return pipe def generate_image(prompt, style): if not prompt.strip(): raise gr.Error("ছবির বর্ণনা লিখুন") model = load_model() full_prompt = f"{prompt}, {style}, detailed, beautiful image" image = model( full_prompt, num_inference_steps=15, guidance_scale=7 ).images[0] return image demo = gr.Interface( fn=generate_image, inputs=[ gr.Textbox(label="Describe Your Image"), gr.Dropdown( ["Realistic", "3D Animation", "Cartoon", "Anime", "Cinematic"], label="Image Style" ) ], outputs=gr.Image(label="Generated Image", type="pil"), title="Free AI Image Studio", description="Create AI images for free!" ) if __name__ == "__main__": demo.launch()
